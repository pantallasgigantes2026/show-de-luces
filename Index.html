    <h1>¡BIENVENIDO AL SHOW!</h1>

    <p>Presiona el botón de abajo para activar tu flash al ritmo de la música del estadio.</p>

    

    <button class="btn" id="startBtn">UNIRSE AL SHOW</button>

    <div class="status" id="statusText">Esperando activación...</div>


    <script>

        const startBtn = document.getElementById('startBtn');

        const statusText = document.getElementById('statusText');

        

        let audioContext;

        let analyser;

        let microphone;

        let javascriptNode;

        let track; // Guardará la pista de video de la cámara para controlar el flash


        // Umbral de sensibilidad (Ajusta este número entre 0 y 255 si el estadio es muy ruidoso)

        const UMBRAL_RITMO = 45; 


        startBtn.addEventListener('click', async () => {

            statusText.innerText = "Solicitando permisos de cámara y micrófono...";

            

            try {

                // 1. Pedir acceso a la cámara trasera para controlar el Flash (torch)

                const videoStream = await navigator.mediaDevices.getUserMedia({

                    video: { facingMode: { exact: "environment" } }

                }).catch(async () => {

                    // Si falla la trasera (ej. en laptops o algunos dispositivos), intenta con cualquier cámara

                    return await navigator.mediaDevices.getUserMedia({ video: true });

                });


                track = videoStream.getVideoTracks()[0];


                // 2. Pedir acceso al micrófono para capturar el audio ambiental del estadio

                const audioStream = await navigator.mediaDevices.getUserMedia({ audio: true });

                

                // Configurar el analizador de audio de JavaScript

                audioContext = new (window.AudioContext || window.webkitAudioContext)();

                analyser = audioContext.createAnalyser();

                microphone = audioContext.createMediaStreamSource(audioStream);

                javascriptNode = audioContext.createScriptProcessor(2048, 1, 1);


                analyser.smoothingTimeConstant = 0.3;

                analyser.fftSize = 1024;


                microphone.connect(analyser);

                analyser.connect(javascriptNode);

                javascriptNode.connect(audioContext.destination);


                // Ocultar botón y cambiar interfaz

                startBtn.style.display = 'none';

                document.body.classList.add('active-glow');

                statusText.innerText = "¡Sincronizado! Mantén tu pantalla encendida y apunta tu flash hacia el frente.";


                // Procesar el audio en tiempo real

                javascriptNode.onaudioprocess = () => {

                    const array = new Uint8Array(analyser.frequencyBinCount);

                    analyser.getByteFrequencyData(array);

                    

                    // Obtener el promedio de volumen de las frecuencias bajas (los golpes/beats de la música)

                    let values = 0;

                    const length = array.length;

                    for (let i = 0; i < length; i++) {

                        values += array[i];

                    }

                    const promedioVolumen = values / length;


                    // Si el volumen supera nuestro umbral, encendemos el flash temporalmente

                    if (promedioVolumen > UMBRAL_RITMO) {

                        controlarFlash(true);

                    } else {

                        controlarFlash(false);

                    }

                };


            } catch (err) {

                console.error(err);

                statusText.innerText = "Error: Asegúrate de aceptar todos los permisos de cámara y micrófono.";

                statusText.style.color = "#ff4444";

            }

        });


        // Función para encender o apagar el flash (Soporta Android y navegadores modernos)

        function controlarFlash(encender) {

            if (track && typeof track.getCapabilities === 'function') {

                const capabilities = track.getCapabilities();

                // Verificar si el dispositivo físicamente tiene flash/antorcha

                if (capabilities.torch) {

                    track.applyConstraints({

                        advanced: [{ torch: encender }]

                    }).catch(e => console.log("Error aplicando flash:", e));

                }

            }

        }
